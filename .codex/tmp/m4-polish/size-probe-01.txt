using System.Text;
using Module.Fish.Data;
using UnityEditor;
using UnityEngine;

public class FishSizeProbe
{
    public static void Run()
    {
        StringBuilder sb = new StringBuilder();
        string[] guids = AssetDatabase.FindAssets("t:Prefab", new[] { "Assets/Prefabs/Fish" });

        foreach (string guid in guids)
        {
            string path = AssetDatabase.GUIDToAssetPath(guid);
            string name = System.IO.Path.GetFileNameWithoutExtension(path);
            FishDataSO data = AssetDatabase.LoadAssetAtPath<FishDataSO>("Assets/Data/Fish/" + name + ".asset");
            GameObject prefab = AssetDatabase.LoadAssetAtPath<GameObject>(path);

            if (prefab == null)
                continue;

            GameObject temp = Object.Instantiate(prefab);
            temp.transform.position = Vector3.zero;
            temp.transform.rotation = Quaternion.identity;

            Renderer r = temp.GetComponentInChildren<Renderer>();
            Vector3 raw = r != null ? r.bounds.size : Vector3.zero;
            Object.DestroyImmediate(temp);

            float display = data != null ? data.DisplayLength : 0f;
            float modelLength = raw.z > 0f ? raw.z : 1f;
            float scale = display > 0f ? display / modelLength : 1f;

            sb.AppendLine(name + " raw(y,z)=" + raw.y.ToString("0.000") + "," + raw.z.ToString("0.000")
                + " display=" + display.ToString("0.00")
                + " -> height=" + (raw.y * scale).ToString("0.000")
                + " cap=" + (data != null ? data.CarryCost : -1)
                + " rarity=" + (data != null ? data.Rarity.ToString() : "-")
                + " valueMul=" + (data != null ? data.ResolveValueMultiplier(null).ToString("0.00") : "-"));
        }

        Debug.Log("[FishSizeProbe] " + sb);
    }
}
