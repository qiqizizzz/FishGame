using UnityEditor;
using UnityEngine;

public class ResultPanelLabels
{
    private const string PANEL_PATH = "Assets/Prefabs/UI/UIResultPanel.prefab";

    public static void Run()
    {
        GameObject root = PrefabUtility.LoadPrefabContents(PANEL_PATH);

        if (root == null)
        {
            Debug.LogError("[ResultPanelLabels] 面板加载失败");
            return;
        }

        setText(root, "Bg/Txt_Title", "本次收获");
        setText(root, "Bg/Txt_Summary", "下潜深度 0 米　·　捕获 0 / 10");
        setText(root, "Bg/Btns/Btn_Again/Text", "继续钓鱼");
        setText(root, "Bg/Btns/Btn_Menu/Text", "退出");

        PrefabUtility.SaveAsPrefabAsset(root, PANEL_PATH);
        PrefabUtility.UnloadPrefabContents(root);
        AssetDatabase.SaveAssets();
        Debug.Log("[ResultPanelLabels] 文案已对齐录屏：本次收获 / 继续钓鱼 / 退出");
    }

    private static void setText(GameObject root, string path, string content)
    {
        Transform node = root.transform.Find(path);

        if (node == null)
        {
            Debug.LogWarning("[ResultPanelLabels] 节点缺失：" + path);
            return;
        }

        TMPro.TMP_Text text = node.GetComponent<TMPro.TMP_Text>();

        if (text == null)
        {
            Debug.LogWarning("[ResultPanelLabels] 无 TMP：" + path);
            return;
        }

        text.text = content;
    }
}
