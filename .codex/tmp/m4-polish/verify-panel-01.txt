using System.Text;
using Module.Fish.Model;
using Module.Fish.Type;
using TMPro;
using UI.Fishing;
using UnityEditor;
using UnityEngine;
using UnityEngine.UI;

public class ResultPanelVerify
{
    private const string PANEL_PATH = "Assets/Prefabs/UI/UIResultPanel.prefab";

    private static StringBuilder s_log = new StringBuilder();
    private static int s_failed;

    public static void Run()
    {
        s_log = new StringBuilder();
        s_failed = 0;

        GameObject prefab = AssetDatabase.LoadAssetAtPath<GameObject>(PANEL_PATH);

        if (prefab == null)
        {
            Debug.LogError("[ResultPanelVerify] prefab 未加载：" + PANEL_PATH);
            return;
        }

        GameObject instance = Object.Instantiate(prefab);
        instance.hideFlags = HideFlags.DontSave;
        Transform root = instance.transform;

        string[] textPaths = { "Bg", "Bg/Txt_Title", "Bg/Txt_Summary", "Bg/Txt_Coins",
            "Bg/Content_Rows", "Bg/Content_Rows/Row_Template", "Bg/Btns",
            "Bg/Btns/Btn_Again", "Bg/Btns/Btn_Menu" };

        foreach (string path in textPaths)
            check(root.Find(path) != null, "节点存在 " + path, "");

        check(root.GetComponent<UIResultPanel>() != null, "root 挂 UIResultPanel", "");
        check(root.GetComponent<Image>() == null, "root 上没有抢色 Image", "");

        Transform template = root.Find("Bg/Content_Rows/Row_Template");
        check(template != null && !template.gameObject.activeSelf, "模板默认隐藏", "");
        check(template != null && template.GetComponent<UIResultFishItem>() != null, "模板挂 UIResultFishItem", "");

        Transform content = root.Find("Bg/Content_Rows");
        VerticalLayoutGroup layout = content != null ? content.GetComponent<VerticalLayoutGroup>() : null;
        check(layout != null && !layout.childControlWidth && !layout.childControlHeight, "行容器布局不拉伸子项",
              layout == null ? "missing" : "controlW=" + layout.childControlWidth + " controlH=" + layout.childControlHeight);

        Button again = root.Find("Bg/Btns/Btn_Again").GetComponent<Button>();
        Button menu = root.Find("Bg/Btns/Btn_Menu").GetComponent<Button>();
        check(again != null && menu != null, "两个按钮都有 Button", "");
        check(again != null && again.targetGraphic != null && again.targetGraphic.raycastTarget, "按钮可点", "");

        int blocked = 0;
        foreach (Graphic graphic in instance.GetComponentsInChildren<Graphic>(true))
            if (graphic.raycastTarget && graphic.name != "Bg" && graphic.name != "Btn_Again" && graphic.name != "Btn_Menu")
                blocked++;

        check(blocked == 0, "只有底板与按钮吃射线", "多余数量=" + blocked);

        UIResultFishItem item = Object.Instantiate(template.GetComponent<UIResultFishItem>(), content);
        item.gameObject.SetActive(true);
        item.Setup(new FishScoreItem
        {
            Id = "clown",
            DisplayName = "小丑鱼",
            Rarity = FishRarity.Common,
            Stars = 2,
            Coins = 27,
            CatchDepth = 12f,
            PrimaryColor = new Color(0.9f, 0.5f, 0.1f, 1f)
        });

        TMP_Text[] texts = item.GetComponentsInChildren<TMP_Text>(true);
        string shown = string.Join(" | ", System.Array.ConvertAll(texts, t => t.text));
        check(shown.Contains("小丑鱼") && shown.Contains("★★") && shown.Contains("+27"), "行模板填数据成功", shown);

        Image mark = item.transform.Find("Img_Mark").GetComponent<Image>();
        check(Mathf.Abs(mark.color.r - 0.9f) < 0.01f, "色标取到档位主色", mark.color.ToString());

        Object.DestroyImmediate(item.gameObject);
        Object.DestroyImmediate(instance);

        s_log.AppendLine(s_failed == 0 ? "ALL PASS" : s_failed + " FAILED");
        Debug.Log("[ResultPanelVerify] " + s_log);
    }

    private static void check(bool ok, string label, string detail)
    {
        if (!ok) s_failed++;
        s_log.AppendLine((ok ? "PASS  " : "FAIL  ") + label + (string.IsNullOrEmpty(detail) ? string.Empty : " :: " + detail));
    }
}
