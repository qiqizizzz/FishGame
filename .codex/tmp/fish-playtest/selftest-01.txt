public class TestHook : IHookRuntime
{
    public HookConfigSO Config;
    public FishingSessionConfigSO Session;
    public IFishingStats Stats { get; set; }
    public IFishingEncounter Encounter { get; set; }
    public HookStateMachine Machine;

    private readonly List<string> m_transitions = new List<string>();
    private float m_x;
    private float m_depth;

    public HookConfigSO HookConfig { get { return Config; } }
    public float ShaftDepth { get { return Session.ShaftDepth; } }
    public float ShaftHalfWidth { get { return Session.ShaftHalfWidth; } }
    public float MaxDepth { get { return Mathf.Min(ShaftDepth, Stats.LineLength); } }
    public float Depth { get { return m_depth; } }
    public float Horizontal { get { return m_x; } }
    public Vector3 HolePosition { get { return new Vector3(0f, 0f, -1.5f); } }
    public float LateralVelocity { get; set; }
    public float LateralInput { get; set; }
    public bool IsPressing { get; set; }

    public void MoveTo(float x, float depth) { m_x = x; m_depth = depth; }

    public void TransitionTo(HookPhase phase)
    {
        m_transitions.Add(phase.ToString());
        Machine.Change(phase, this);
    }

    public string Transitions { get { return string.Join(">", m_transitions); } }
}

public class FishPlaySelfTest
{
    private const float MAX_SWAY = 0.55f;
    private const float HEIGHT_RATIO = 0.35f;

    private static StringBuilder s_log = new StringBuilder();
    private static int s_failed;

    private static void Check(string label, bool ok, string detail)
    {
        s_log.AppendLine((ok ? "PASS  " : "FAIL  ") + label + " :: " + detail);
        if (!ok) s_failed++;
    }

    public static void Run()
    {
        s_log = new StringBuilder();
        s_failed = 0;

        HookConfigSO config = AssetDatabase.LoadAssetAtPath<HookConfigSO>("Assets/Data/Config/HookConfig.asset");
        FishingSessionConfigSO session = AssetDatabase.LoadAssetAtPath<FishingSessionConfigSO>("Assets/Data/Config/FishingSession_Island1.asset");
        RarityTableSO table = AssetDatabase.LoadAssetAtPath<RarityTableSO>("Assets/Data/Config/RarityTable.asset");
        FishDataSO clown = AssetDatabase.LoadAssetAtPath<FishDataSO>("Assets/Data/Fish/Clownfish.asset");
        FishDataSO tuna = AssetDatabase.LoadAssetAtPath<FishDataSO>("Assets/Data/Fish/Tuna.asset");
        FishDataSO tiny = AssetDatabase.LoadAssetAtPath<FishDataSO>("Assets/Data/Fish/SuperdwarfFish.asset");
        FishDataSO bass = AssetDatabase.LoadAssetAtPath<FishDataSO>("Assets/Data/Fish/Bass.asset");
        FishDataSO puffer = AssetDatabase.LoadAssetAtPath<FishDataSO>("Assets/Data/Fish/Pufferfish.asset");

        List<FishInstance> fishes = new List<FishInstance>();
        List<GameObject> trash = new List<GameObject>();

        testHookStateMachine(config, session);
        testContactRules(config, clown, puffer, fishes, trash);
        testCarryStack(config, clown, tuna, tiny, bass, table, fishes, trash);
        testSpotAvoidance(clown, fishes, trash);
        testSessionResult();

        for (int i = 0; i < trash.Count; i++)
            Object.DestroyImmediate(trash[i]);

        s_log.AppendLine(s_failed == 0 ? "ALL PASS" : s_failed + " FAILED");
        Debug.Log("[FishPlaySelfTest]\n" + s_log);
    }

    private static void testHookStateMachine(HookConfigSO config, FishingSessionConfigSO session)
    {
        TestHook hook = new TestHook
        {
            Config = config,
            Session = session,
            Stats = new DefaultFishingStats(config)
        };

        HookStateMachine machine = new HookStateMachine(
            new IdleHookState(), new DescendingHookState(), new ReelingHookState(),
            new TangledHookState(), new LandedHookState());
        hook.Machine = machine;
        machine.Start(hook);

        const float DT = 1f / 60f;

        hook.IsPressing = true;
        hook.LateralInput = 0.06f;
        for (int i = 0; i < 60; i++) machine.Update(hook, DT);
        Check("下潜 1 秒", hook.Depth > 5f && hook.Depth < 7f, "depth=" + hook.Depth.ToString("0.00"));
        Check("下潜可横移", Mathf.Abs(hook.Horizontal) > 0.05f, "x=" + hook.Horizontal.ToString("0.000"));

        hook.IsPressing = false;
        float xBefore = hook.Horizontal;
        hook.LateralInput = -0.08f;
        float depthBefore = hook.Depth;
        for (int i = 0; i < 30; i++) machine.Update(hook, DT);
        Check("松手转回收", hook.Transitions.Contains("Reeling"), hook.Transitions);
        Check("回收速度已提", depthBefore - hook.Depth > 2.0f, "0.5 秒收了 " + (depthBefore - hook.Depth).ToString("0.00") + " 米");
        Check("回收仍可横移", Mathf.Abs(hook.Horizontal - xBefore) > 0.05f,
              "x " + xBefore.ToString("0.000") + " -> " + hook.Horizontal.ToString("0.000"));
        Check("回收横移夹在井道内", Mathf.Abs(hook.Horizontal) <= session.ShaftHalfWidth + 0.001f, "x=" + hook.Horizontal.ToString("0.000"));

        hook.LateralInput = 0f;
        hook.LateralVelocity = 0f;
        float xStatic = hook.Horizontal;
        for (int i = 0; i < 10; i++) machine.Update(hook, DT);
        Check("无输入时不漂移", Mathf.Abs(hook.Horizontal - xStatic) < 0.001f, "x=" + hook.Horizontal.ToString("0.000"));

        for (int i = 0; i < 600 && machine.Phase != HookPhase.Landed; i++)
            machine.Update(hook, DT);
        Check("到底自动出水", machine.Phase == HookPhase.Landed, "phase=" + machine.Phase);
    }

    private static void testContactRules(HookConfigSO config, FishDataSO clown, FishDataSO puffer,
        List<FishInstance> fishes, List<GameObject> trash)
    {
        FishDetectionService detection = new FishDetectionService();
        FishAiService ai = new FishAiService();

        FishInstance headRight = makeFish(clown, 1.0f, 10f, 1, trash, fishes);
        FishInstance pufferFish = makeFish(puffer, 0f, 14f, -1, trash, fishes);
        detection.Rebuild(fishes);

        float headX = headRight.X + headRight.Direction * clown.DisplayLength * 0.5f;
        FishInstance found;
        Check("蹭到鱼头算捕获", detection.TryFindTouched(headX, 10f, config.HookRadius, true, out found) && found == headRight, "");

        FishInstance tailFound;
        float tailX = headRight.X - headRight.Direction * clown.DisplayLength * 0.5f;
        Check("只认鱼头：尾部不触发", !detection.TryFindTouched(tailX, 10f, config.HookRadius, true, out tailFound), "");
        Check("鱼身任意接触可检出", detection.TryFindTouched(headRight.X, 10f, config.HookRadius, false, out tailFound), "");

        FishInstance hazardFound;
        Check("河豚可检出", detection.TryFindTouched(0f, 14f, config.HookRadius, false, out hazardFound) && hazardFound == pufferFish, "");

        FishInstance victim = fishes[0];
        victim.X = -0.4f;
        ai.Scatter(victim, 0.4f);
        Check("惊散方向背离钩子", victim.Direction < 0 && victim.FleeTimer > 0f, "dir=" + victim.Direction);
        float fleeFrom = victim.X;
        for (int i = 0; i < 30; i++) ai.Advance(fishes, 1f / 60f, 2.83f);
        Check("惊散位移明显", victim.X < fleeFrom - 0.3f, "x " + fleeFrom.ToString("0.00") + " -> " + victim.X.ToString("0.00"));
    }

    private static void testCarryStack(HookConfigSO config, FishDataSO clown, FishDataSO tuna, FishDataSO tiny,
        FishDataSO bass, RarityTableSO table, List<FishInstance> fishes, List<GameObject> trash)
    {
        FishCarryService carry = new FishCarryService();
        carry.BeginSession(new DefaultFishingStats(config), table);

        int attached = 0;
        for (int i = 0; i < 8; i++)
        {
            FishInstance fake = makeFish(clown, 2f + i, 10f, 1, trash, fishes);
            if (carry.Attach(fake) != null) attached++;
        }
        Check("携带容量封顶", attached > 0 && carry.UsedPoints <= config.CarryCapacity,
              "挂了 " + carry.Count + " 条 / 容量 " + config.CarryCapacity + " 点");

        FishDetectionService detection = new FishDetectionService();
        detection.Rebuild(fishes);
        FishInstance hookedFound;
        Check("钩上的鱼不再参与判定", !detection.TryFindTouched(2f, 10f, config.HookRadius, false, out hookedFound), "");

        carry.Follow(0.5f, 6f);
        CatchRecord nearest = carry.Carried[0];
        Check("第一条鱼贴在钩下", nearest.Fish.Depth > 6f && Mathf.Abs(nearest.Fish.X - 0.5f) <= MAX_SWAY + 0.001f,
              "depth=" + nearest.Fish.Depth.ToString("0.00") + " x=" + nearest.Fish.X.ToString("0.00"));

        List<CatchRecord> released = new List<CatchRecord>();
        carry.DetachByChance(1f, released);
        Check("受罚全脱钩", released.Count == attached && carry.Count == 0 && carry.UsedPoints == 0,
              "脱 " + released.Count + " 条，剩余 " + carry.Count);

        // 混合体型：大 tuna 与小 superdwarf 同串，必须按各自身高留位
        FishCarryService stacked = new FishCarryService();
        stacked.BeginSession(new TestStats(), table);

        FishInstance f1 = makeFish(clown, 0f, 12f, 1, trash, fishes);
        FishInstance f2 = makeFish(tuna, 0f, 12f, -1, trash, fishes);
        FishInstance f3 = makeFish(tiny, 0f, 12f, 1, trash, fishes);
        FishInstance f4 = makeFish(bass, 0f, 12f, -1, trash, fishes);

        int stackedCount = 0;
        FishInstance[] group = { f1, f2, f3, f4 };
        for (int i = 0; i < group.Length; i++)
            if (stacked.Attach(group[i]) != null) stackedCount++;

        stacked.Follow(0f, 10f);
        Check("四条混合体型的鱼都挂上了", stackedCount == 4 && stacked.Count == 4, "实际 " + stacked.Count);

        bool monotonic = true;
        bool gapOk = true;
        bool swayOk = true;
        float lastSwaySign = 0f;

        for (int i = 0; i < stacked.Count; i++)
        {
            FishInstance fish = stacked.Carried[i].Fish;

            if (fish.Depth <= 10f) monotonic = false;

            float sway = fish.X;
            float swayLimit = Mathf.Min(fish.Data.DisplayLength * 0.34f, MAX_SWAY);

            if (Mathf.Abs(sway) > swayLimit + 0.001f) swayOk = false;

            float sign = Mathf.Sign(sway);
            if (i > 0 && sign == lastSwaySign) swayOk = false;
            lastSwaySign = sign;

            if (i == 0) continue;

            FishInstance previous = stacked.Carried[i - 1].Fish;

            if (fish.Depth <= previous.Depth) monotonic = false;

            float minGap = (previous.DisplayHeight + fish.DisplayHeight) * 0.5f;
            float gap = fish.Depth - previous.Depth;

            if (gap < minGap) gapOk = false;

            if (i == 1)
                Check("大鱼小鱼之间留出间隙", gapOk,
                      "tuna-clown gap=" + gap.ToString("0.000") + " 需要>=" + minGap.ToString("0.000")
                      + "（旧固定间距 0.5 会叠）");
        }

        Check("鱼串自上而下不重叠", monotonic && gapOk, "顺序递增=" + monotonic + " 间隙达标=" + gapOk);
        Check("鱼串左右交替错位", swayOk, "sway=" + f1.X.ToString("0.00") + "," + f2.X.ToString("0.00")
              + "," + f3.X.ToString("0.00") + "," + f4.X.ToString("0.00"));
        Check("挂上钩的鱼带下垂角", !Mathf.Approximately(f1.Root.localRotation.z, f2.Root.localRotation.z),
              "z1=" + f1.Root.localRotation.z.ToString("0.0") + " z2=" + f2.Root.localRotation.z.ToString("0.0"));

        List<CatchRecord> landed = new List<CatchRecord>();
        stacked.Unload(landed);
        FishInstance prize = makeFish(clown, 0f, 5f, 1, trash, fishes);
        carry.Attach(prize);
        carry.Unload(landed);
        Check("出水结算清空负载", landed.Count == 5 && carry.Count == 0, "landed=" + landed.Count);
    }

    private static void testSpotAvoidance(FishDataSO clown, List<FishInstance> fishes, List<GameObject> trash)
    {
        FishSpawnService spawn = new FishSpawnService();
        FieldInfo aliveField = typeof(FishSpawnService).GetField("m_alive", BindingFlags.NonPublic | BindingFlags.Instance);
        MethodInfo isSpotFree = typeof(FishSpawnService).GetMethod("isSpotFree", BindingFlags.NonPublic | BindingFlags.Instance);

        if (aliveField == null || isSpotFree == null)
        {
            Check("落点避让可测", false, "m_alive/isSpotFree 反射未找到");
            return;
        }

        IList alive = (IList)aliveField.GetValue(spawn);
        float length = clown.DisplayLength;
        float height = clown.DisplayLength * HEIGHT_RATIO;

        Check("空场可落点", (bool)isSpotFree.Invoke(spawn, new object[] { 0f, 10f, length, height }), "");

        FishInstance neighbor = makeFish(clown, 0f, 10f, 1, trash, fishes);
        neighbor.DisplayHeight = height;
        alive.Add(neighbor);

        Check("原位重迭被拒", !(bool)isSpotFree.Invoke(spawn, new object[] { 0f, 10f, length, height }), "");
        Check("横向拉开后可落点", (bool)isSpotFree.Invoke(spawn, new object[] { 2f, 10f, length, height }), "");
        Check("纵深拉开后可落点", (bool)isSpotFree.Invoke(spawn, new object[] { 0f, 13f, length, height }), "");
        Check("钩上的鱼不挡落点", (bool)isSpotFree.Invoke(spawn, new object[] { 0f, 10f, length, height }) == false, "仍占用属预期");

        neighbor.IsHooked = true;
        Check("已上钩的鱼让位", (bool)isSpotFree.Invoke(spawn, new object[] { 0f, 10f, length, height }), "");
        alive.Clear();
    }

    private static void testSessionResult()
    {
        SessionResult result = new SessionResult(10);
        result.Add(new FishScoreItem { Id = "a", DisplayName = "A", Rarity = FishRarity.Common, Stars = 1, Coins = 12, CatchDepth = 3f });
        result.Add(new FishScoreItem { Id = "b", DisplayName = "B", Rarity = FishRarity.Rare, Stars = 3, Coins = 30, CatchDepth = 21f });

        Check("结算累计金币与条数", result.TotalCoins == 42 && result.Items.Count == 2 && !result.IsTargetReached,
              "coins=" + result.TotalCoins + " items=" + result.Items.Count);

        result.Finish(2, 18.4f);
        Check("封盘写入总数与最大深度", result.TotalCatch == 2 && Mathf.Abs(result.MaxDepth - 18.4f) < 0.001f,
              "total=" + result.TotalCatch + " maxDepth=" + result.MaxDepth.ToString("0.0"));

        SessionResult full = new SessionResult(2);
        full.Add(new FishScoreItem { Coins = 1 });
        full.Add(new FishScoreItem { Coins = 1 });
        full.Finish(2, 5f);
        Check("达标判定", full.IsTargetReached, "total=" + full.TotalCatch);
    }

    private static FishInstance makeFish(FishDataSO data, float x, float depth, int direction,
        List<GameObject> trash, List<FishInstance> fishes)
    {
        GameObject go = new GameObject("test_fish");
        go.hideFlags = HideFlags.DontSave;
        go.transform.localPosition = new Vector3(x, -depth, 0f);
        trash.Add(go);

        FishInstance fish = new FishInstance
        {
            Data = data,
            View = go,
            Root = go.transform,
            Behavior = data.Behavior,
            BaseDepth = depth,
            DisplayHeight = data.DisplayLength * HEIGHT_RATIO,
            X = x,
            Depth = depth,
            Direction = direction,
            Speed = 1f,
            Phase = 0f
        };

        fishes.Add(fish);
        return fish;
    }
