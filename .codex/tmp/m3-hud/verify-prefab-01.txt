using System.Text;
using TMPro;
using UI.Fishing;
using UnityEditor;
using UnityEngine;
using UnityEngine.U2D;
using UnityEngine.UI;

public class HudPrefabVerify
{
    private const string PREFAB_PATH = "Assets/Prefabs/UI/UIFishingHUD.prefab";

    public static void Run()
    {
        StringBuilder log = new StringBuilder();
        int failed = 0;

        GameObject prefab = AssetDatabase.LoadAssetAtPath<GameObject>(PREFAB_PATH);

        if (prefab == null)
        {
            Debug.LogError("[HudPrefabVerify] prefab 未加载：" + PREFAB_PATH);
            return;
        }

        GameObject instance = (GameObject)Object.Instantiate(prefab);
        instance.hideFlags = HideFlags.DontSave;

        RectTransform root = (RectTransform)instance.transform;
        check(log, ref failed, "root 全屏拉伸",
            root.anchorMin == Vector2.zero && root.anchorMax == Vector2.one,
            "anchorMin=" + root.anchorMin + " anchorMax=" + root.anchorMax);
        check(log, ref failed, "root 挂 UIFishingHUD", instance.GetComponent<UIFishingHUD>() != null, "");
        check(log, ref failed, "Panel_Counters 有 Image",
            instance.GetComponent<Image>() == null && root.Find("Panel_Counters").GetComponent<Image>() != null,
            "HUD 只从 Panel_Counters 取 Image，root 上不能有 Image 抢色");

        string[] rows = { "Row_Hook", "Row_Line", "Row_Load" };
        string[] labels = { "鱼钩", "鱼线", "负载" };

        for (int i = 0; i < rows.Length; i++)
        {
            string path = "Panel_Counters/" + rows[i];
            Transform row = root.Find(path);
            bool ok = row != null
                && row.GetComponent<UIHudCounterItem>() != null
                && row.Find("Txt_Label") != null && row.Find("Txt_Label").GetComponent<TMP_Text>() != null
                && row.Find("Txt_Value") != null && row.Find("Txt_Value").GetComponent<TMP_Text>() != null;

            check(log, ref failed, path + " 结构与 TMP 齐全", ok,
                row == null ? "missing" : "label=" + labelOf(row) + " value=" + valueOf(row));

            if (row != null && row.GetComponent<UIHudCounterItem>() != null)
            {
                string labelText = labelOf(row);
                check(log, ref failed, rows[i] + " 标签文案= " + labels[i], labelText == labels[i], "实际=" + labelText);
            }
        }

        Transform phase = root.Find("Txt_Phase");
        check(log, ref failed, "Txt_Phase 存在且有 TMP",
            phase != null && phase.GetComponent<TMP_Text>() != null, phase == null ? "missing" : "");

        int raycastOn = 0;
        foreach (Graphic graphic in instance.GetComponentsInChildren<Graphic>(true))
            if (graphic.raycastTarget) raycastOn++;

        check(log, ref failed, "全部 Graphics 不拦射线", raycastOn == 0, "raycastTarget=true 的数量=" + raycastOn);

        int missingFont = 0;
        foreach (TMP_Text text in instance.GetComponentsInChildren<TMP_Text>(true))
            if (text.font == null) missingFont++;

        check(log, ref failed, "全部 TMP 已绑字体", missingFont == 0, "缺字体数量=" + missingFont);

        Object.DestroyImmediate(instance);

        log.AppendLine(failed == 0 ? "ALL PASS" : failed + " FAILED");
        Debug.Log("[HudPrefabVerify] " + log);
    }

    private static string labelOf(Transform row)
    {
        Transform node = row.Find("Txt_Label");
        return node == null ? "-" : node.GetComponent<TMP_Text>().text;
    }

    private static string valueOf(Transform row)
    {
        Transform node = row.Find("Txt_Value");
        return node == null ? "-" : node.GetComponent<TMP_Text>().text;
    }

    private static void check(StringBuilder log, ref int failed, string label, bool ok, string detail)
    {
        if (!ok) failed++;
        log.AppendLine((ok ? "PASS  " : "FAIL  ") + label + (string.IsNullOrEmpty(detail) ? string.Empty : " :: " + detail));
    }
}
