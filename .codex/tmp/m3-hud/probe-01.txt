using System;
using System.Text;
using UnityEditor;
using UnityEngine;

public class HudProbe
{
    public static void Run()
    {
        StringBuilder sb = new StringBuilder();
        Type hud = Type.GetType("UI.Fishing.UIFishingHUD, Assembly-CSharp");
        Type item = Type.GetType("UI.Fishing.UIHudCounterItem, Assembly-CSharp");
        Type ctrl = Type.GetType("Module.Fish.FishingController, Assembly-CSharp");

        sb.AppendLine("UIFishingHUD=" + (hud != null) + " base=" + (hud == null ? "-" : hud.BaseType.Name));
        sb.AppendLine("UIHudCounterItem=" + (item != null) + " base=" + (item == null ? "-" : item.BaseType.Name));

        string[] props = { "Depth", "CarriedPoints", "CarryCapacity", "TargetCatchCount", "TotalCatch" };
        foreach (string name in props)
            sb.AppendLine("FishingController." + name + "=" + (ctrl != null && ctrl.GetProperty(name) != null));

        GameObject prefab = AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Prefabs/UI/UIFishingHUD.prefab");
        sb.AppendLine("prefabLoaded=" + (prefab != null));
        sb.AppendLine("hudComponentOnPrefab=" + (prefab != null && prefab.GetComponent(hud) != null));

        Type res = Type.GetType("Common.Res.ResManager, Assembly-CSharp");
        sb.AppendLine("ResManager found=" + (res != null));

        Debug.Log("[HudProbe] " + sb);
    }
}
