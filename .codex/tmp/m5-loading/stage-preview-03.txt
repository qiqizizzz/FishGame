using UnityEngine;
using UnityEngine.UI;

public class LoadingPanelReStage
{
    public static void Run()
    {
        GameObject canvasObject = GameObject.Find("PreviewCanvas");

        if (canvasObject == null)
        {
            Debug.LogError("[LoadingPanelReStage] 未找到 PreviewCanvas");
            return;
        }

        Camera camera = GameObject.Find("PreviewCamera")?.GetComponent<Camera>();
        Canvas canvas = canvasObject.GetComponent<Canvas>();
        canvas.renderMode = RenderMode.ScreenSpaceCamera;
        canvas.worldCamera = camera;
        canvas.planeDistance = 1f;

        Transform panel = canvasObject.transform.Find("UILoadingPanel");
        Canvas nested = panel == null ? null : panel.GetComponent<Canvas>();

        if (nested != null)
            Object.DestroyImmediate(nested);

        Debug.Log("[LoadingPanelReStage] canvas=" + canvas.renderMode
            + " nestedRemoved=" + (nested != null) + " camera=" + (camera == null ? "NULL" : camera.name));
    }
}
