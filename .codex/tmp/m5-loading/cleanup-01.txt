using System.Text;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.SceneManagement;

public class LoadingPanelCleanup
{
    public static void Run()
    {
        StringBuilder sb = new StringBuilder();

        for (int i = SceneManager.sceneCount - 1; i >= 0; i--)
        {
            Scene scene = SceneManager.GetSceneAt(i);
            sb.AppendLine("scene[" + i + "] name='" + scene.name + "' path='" + scene.path + "' dirty=" + scene.isDirty);

            if (string.IsNullOrEmpty(scene.path))
            {
                EditorSceneManager.CloseScene(scene, true);
                sb.AppendLine("  -> closed");
            }
        }

        Scene game = EditorSceneManager.GetSceneByName("Game");

        if (game.IsValid())
            EditorSceneManager.SetActiveScene(game);

        Debug.Log("[LoadingPanelCleanup]\n" + sb);
    }
}
