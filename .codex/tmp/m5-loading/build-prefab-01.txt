using System.Collections.Generic;
using System.Text;
using TMPro;
using UI.Common;
using UnityEditor;
using UnityEditor.AddressableAssets;
using UnityEditor.AddressableAssets.Settings;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.SceneManagement;
using UnityEngine.UI;

public class LoadingPrefabBuilder
{
    private const string PREFAB_PATH = "Assets/Prefabs/UI/UILoadingPanel.prefab";
    private const string FONT_PATH = "Assets/Font/MaoKenTangYuan-beta0.12 SDF.asset";
    private const string RING_PATH = "Assets/Art/UI/Common/Img_Loading_Ring.png";
    private const string ADDRESS = "UILoadingPanel";
    private const string GROUP_NAME = "UI";

    private static readonly Color RingColor = new Color(0.92f, 0.96f, 1f, 1f);
    private static readonly Color TextColor = new Color(0.75f, 0.82f, 0.9f, 1f);

    private static StringBuilder s_log = new StringBuilder();

    public static void Run()
    {
        s_log = new StringBuilder();
        TMP_FontAsset font = AssetDatabase.LoadAssetAtPath<TMP_FontAsset>(FONT_PATH);
        Sprite ring = AssetDatabase.LoadAssetAtPath<Sprite>(RING_PATH);

        if (font == null || ring == null)
        {
            Debug.LogError("[LoadingPrefabBuilder] 依赖缺失 font=" + (font == null ? "NULL" : "ok")
                + " ring=" + (ring == null ? "NULL" : "ok"));
            return;
        }

        Scene previous = EditorSceneManager.GetActiveScene();
        Scene temp = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Additive);

        RectTransform root = buildHierarchy(font, ring);
        PrefabUtility.SaveAsPrefabAsset(root.gameObject, PREFAB_PATH, out bool saved);
        s_log.AppendLine("saved=" + saved + " path=" + PREFAB_PATH);

        EditorSceneManager.CloseScene(temp, true);

        if (previous.IsValid())
            EditorSceneManager.SetActiveScene(previous);

        if (saved)
            registerAddressable();

        AssetDatabase.SaveAssets();
        AssetDatabase.Refresh();
        Debug.Log("[LoadingPrefabBuilder]\n" + s_log);
    }

    private static RectTransform buildHierarchy(TMP_FontAsset font, Sprite ring)
    {
        RectTransform root = node("UILoadingPanel", null, Vector2.zero, Vector2.one,
            new Vector2(0.5f, 0.5f), Vector2.zero, Vector2.zero);
        root.gameObject.AddComponent<Canvas>();

        RectTransform black = node("Img_Black", root, Vector2.zero, Vector2.one,
            new Vector2(0.5f, 0.5f), Vector2.zero, Vector2.zero);
        Image mask = black.gameObject.AddComponent<Image>();
        mask.color = new Color(0f, 0f, 0f, 1f);
        mask.raycastTarget = true;

        RectTransform ringRect = node("Img_Ring", root, new Vector2(1f, 0f), new Vector2(1f, 0f),
            new Vector2(0.5f, 0.5f), new Vector2(-100f, 150f), new Vector2(120f, 120f));
        Image ringImage = ringRect.gameObject.AddComponent<Image>();
        ringImage.sprite = ring;
        ringImage.color = RingColor;
        ringImage.type = Image.Type.Simple;
        ringImage.preserveAspect = true;
        ringImage.raycastTarget = false;

        RectTransform text = node("Txt_Loading", root, new Vector2(1f, 0f), new Vector2(1f, 0f),
            new Vector2(1f, 0.5f), new Vector2(-40f, 48f), new Vector2(420f, 56f));
        TextMeshProUGUI label = text.gameObject.AddComponent<TextMeshProUGUI>();
        label.font = font;
        label.text = "加载中...";
        label.fontSize = 40f;
        label.alignment = TextAlignmentOptions.Right;
        label.color = TextColor;
        label.enableWordWrapping = false;
        label.overflowMode = TextOverflowModes.Overflow;
        label.raycastTarget = false;

        root.gameObject.AddComponent<UILoadingPanel>();
        return root;
    }

    private static RectTransform node(string name, RectTransform parent, Vector2 anchorMin, Vector2 anchorMax,
        Vector2 pivot, Vector2 anchoredPosition, Vector2 sizeDelta)
    {
        GameObject go = new GameObject(name, typeof(RectTransform));
        RectTransform rect = (RectTransform)go.transform;

        if (parent != null)
            rect.SetParent(parent, false);

        rect.anchorMin = anchorMin;
        rect.anchorMax = anchorMax;
        rect.pivot = pivot;
        rect.anchoredPosition = anchoredPosition;
        rect.sizeDelta = sizeDelta;
        return rect;
    }

    private static void registerAddressable()
    {
        AddressableAssetSettings settings = AddressableAssetSettingsDefaultObject.Settings;
        AddressableAssetGroup group = settings == null ? null : settings.FindGroup(GROUP_NAME);

        if (group == null)
        {
            s_log.AppendLine("未找到 Addressable 分组 " + GROUP_NAME);
            return;
        }

        foreach (AddressableAssetEntry entry in group.entries)
        {
            if (entry.address == ADDRESS)
                s_log.AppendLine("已存在条目，将重新指向：" + entry.guid);
        }

        string guid = AssetDatabase.AssetPathToGUID(PREFAB_PATH);

        if (string.IsNullOrEmpty(guid))
        {
            s_log.AppendLine("资产 GUID 为空，注册跳过");
            return;
        }

        AddressableAssetEntry created = settings.CreateOrMoveEntry(guid, group, false, false);
        created.address = ADDRESS;
        settings.SetDirty(AddressableAssetSettings.ModificationEvent.EntryCreated, created, true, true);
        s_log.AppendLine("addressable ok address=" + ADDRESS + " group=" + group.name + " guid=" + guid);
    }
}
