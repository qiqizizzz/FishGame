using UnityEngine;
using UnityEngine.UI;

public class LoadingPanelReStage
{
    public static void Run()
    {
        Canvas canvas = Object.FindObjectOfType<Canvas>();

        if (canvas == null)
        {
            Debug.LogError("[LoadingPanelReStage] 未找到 PreviewCanvas");
            return;
        }

        Camera camera = Object.FindObjectOfType<Camera>();
        canvas.renderMode = RenderMode.ScreenSpaceCamera;
        canvas.worldCamera = camera;
        canvas.planeDistance = 1f;
        canvas.sortingOrder = 0;

        Debug.Log("[LoadingPanelReStage] canvas=" + canvas.name + " children=" + canvas.transform.childCount
            + " camera=" + (camera == null ? "NULL" : camera.name));
    }
}
